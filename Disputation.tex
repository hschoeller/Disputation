% WHY: the methodological contribution of P1 in one picture;
%   this is a gap-method slide ("what could you not do
%   before").
% SAY: "The standard way to answer that question is brute
%   force: for every noise level, simulate a long stochastic
%   trajectory and count lifetimes -- expensive, and it
%   requires the model. Our method needs ONE deterministic
%   trajectory. We turn it into a Markov chain: one step of
%   the chain is one deterministic step T followed by one
%   random jump D between points of the trajectory, weighted
%   by distance -- that jump plays the role of the noise. The
%   expected time to escape the regime is then pure linear
%   algebra on that chain -- no re-simulation, and in fact the
%   governing equations never need to be known."
\begin{frame}{One deterministic trajectory suffices to predict
              the noisy regime lifetimes}
  \textbf{Ansatz:} one deterministic step, one random step:
  \begin{equation*}
    P(\sigma) = T \circ D(\sigma), \qquad
    D(\sigma)_{ij} \sim
    \exp\!\left(-\tfrac{\lVert X_i - X_j\rVert^2}
                       {2\sigma\,\dd t}\right)
  \end{equation*}
  \begin{columns}
    \begin{column}{.55\textwidth}
      \begin{itemize}
        \item Expected escape time from regime $\mathcal M$:
              linear algebra on $P(\sigma)$
        \item No stochastic re-simulation; dynamics need not
              be known
      \end{itemize}
    \end{column}
    \begin{column}{.45\textwidth}
      \centering
      % <<< ADAPT: exists in paper-talk Figs/ >>>
      % \includegraphics[width=.45\linewidth]{chain_deterministic.png}
      % \includegraphics[width=.45\linewidth]{chain_stochastic.png}
      \figplaceholder[.9\linewidth]{chain sketches:
        deterministic vs.\ with noise}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Regime lifetimes of the Markov chain}
  \textbf{Ansatz:} one deterministic step $T$, one random step
  $D(\sigma)$:
  \begin{align*}
    P(\sigma) &= T \circ D(\sigma), &
    \left[P(\sigma)\right]_{ij}
      &= \mathbb P\left(x_{n+1} = X_j \mid x_n = X_i\right)
  \end{align*}
  $Q(\sigma)$: restriction of $P(\sigma)$ to $\mathcal M$:
  \begin{align*}
    \frac{1}{\vert I_{\text{entry}}\vert}
    \sum_{i \in I_{\text{entry}}}\sum_{k=0}^\infty\sum_{j}
    \left[Q(\sigma)^k\right]_{ij}\cdot\Delta t
    \quad : \quad \text{expected regime lifetime}
  \end{align*}
\end{frame}
